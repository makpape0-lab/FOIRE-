-- FASECA-KB 2026 : verrouillage de sécurité (à exécuter en une fois dans SQL Editor) begin; -- 1. Aucun rôle d'équipe attribué automatiquement à l'inscription create or replace function public.handle_new_user() returns trigger language plpgsql security definer set search_path to 'public' as $$ begin insert into public.profiles (id, email, nom, telephone, role) values (new.id, lower(new.email), new.raw_user_meta_data->>'nom', new.raw_user_meta_data->>'telephone', 'porteur') on conflict (id) do update set email = excluded.email; return new; end $$; -- 2. Création d'un dossier : règles imposées par le serveur create or replace function public.faseca_before_insert() returns trigger language plpgsql security definer set search_path to 'public' as $$ declare cfg jsonb; maxi int; n int; begin if current_setting('faseca.restore', true) = '1' then return new; end if; perform pg_advisory_xact_lock(hashtext('faseca_inscriptions')); select data into cfg from public.faseca_config where id = 1; if auth.uid() is not null and not public.is_super() then if now() > coalesce(nullif(cfg->>'limite','')::timestamptz, timestamptz '2026-12-10 18:00:00+00') then raise exception 'Les inscriptions sont closes.'; end if; new.user_id := auth.uid(); new.email := lower(coalesce(auth.jwt()->>'email', new.email)); new.created_at := now(); if exists (select 1 from public.faseca_inscriptions where user_id = new.user_id) then raise exception 'Un dossier existe déjà pour ce compte.'; end if; if exists (select 1 from jsonb_array_elements_text(coalesce(to_jsonb(new.photos), '[]'::jsonb)) p where split_part(p, '/', 1) <> new.user_id::text) then raise exception 'Photographie non autorisée.'; end if; end if; if new.formule is null or new.formule not in ('standard','institutionnel') then raise exception 'Formule invalide.'; end if; if length(to_jsonb(new)::text) > 30000 or jsonb_array_length(coalesce(to_jsonb(new.photos), '[]'::jsonb)) > 6 then raise exception 'Dossier trop volumineux.'; end if; maxi := coalesce((cfg->>'stands')::int, 50); select count(*) into n from public.faseca_inscriptions where statut not in ('Refusée','Liste d''attente'); new.numero := 'FASECA-2026-' || lpad(nextval('public.faseca_seq')::text, 4, '0'); new.montant_verse := 0; new.stand := null; new.notes := null; new.statut := case when n >= maxi then 'Liste d''attente' else 'Reçue' end; return new; end $$; -- 3. Modification : l'exposant ne touche ni au statut, ni à l'argent, ni au stand create or replace function public.faseca_before_update() returns trigger language plpgsql security definer set search_path to 'public' as $$ declare tarif numeric; begin if auth.uid() is not null and not public.is_super() then new.id := old.id; new.numero := old.numero; new.user_id := old.user_id; new.email := old.email; new.formule := old.formule; new.statut := old.statut; new.montant_verse := old.montant_verse; new.stand := old.stand; new.notes := old.notes; new.created_at := old.created_at; if to_jsonb(new.photos) is distinct from to_jsonb(old.photos) and ( jsonb_array_length(coalesce(to_jsonb(new.photos), '[]'::jsonb)) > 6 or exists (select 1 from jsonb_array_elements_text(coalesce(to_jsonb(new.photos), '[]'::jsonb)) p where split_part(p, '/', 1) <> old.user_id::text)) then raise exception 'Photographies non autorisées (6 maximum).'; end if; if length(to_jsonb(new)::text) > 30000 then raise exception 'Dossier trop volumineux.'; end if; end if; if new.formule is distinct from old.formule and new.formule not in ('standard','institutionnel') then raise exception 'Formule invalide.'; end if; if new.montant_verse is distinct from old.montant_verse or new.formule is distinct from old.formule then select coalesce((data->>new.formule)::numeric, case new.formule when 'standard' then 50000 else 75000 end) into tarif from (select (select data from public.faseca_config where id = 1) as data) c; if new.montant_verse < 0 or new.montant_verse > tarif then raise exception 'Le montant versé doit être compris entre 0 et %.', tarif; end if; end if; if new.statut is distinct from old.statut and new.statut not in ('Reçue','Réservation confirmée','Stand attribué','Soldée','Liste d''attente','Refusée') then raise exception 'Statut invalide.'; end if; return new; end $$; -- 4. Un seul déclencheur par action (supprime les doublons) drop trigger if exists trg_faseca_before on public.faseca_inscriptions; drop trigger if exists faseca_bi on public.faseca_inscriptions; drop trigger if exists trg_faseca_guard on public.faseca_inscriptions; drop trigger if exists faseca_bu on public.faseca_inscriptions; drop trigger if exists trg_faseca_log on public.faseca_inscriptions; drop trigger if exists faseca_aj on public.faseca_inscriptions; create trigger faseca_bi before insert on public.faseca_inscriptions for each row execute function public.faseca_before_insert(); create trigger faseca_bu before update on public.faseca_inscriptions for each row execute function public.faseca_before_update(); create trigger faseca_aj after insert or update or delete on public.faseca_inscriptions for each row execute function public.faseca_log(); drop function if exists public.faseca_before(); drop function if exists public.faseca_guard(); -- 5. Restauration depuis le journal sans remise à zéro des paiements create or replace function public.faseca_restaurer(p uuid) returns void language plpgsql security definer set search_path to 'public' as $$ declare j public.faseca_journal; begin if not public.is_super() then raise exception 'Accès refusé.'; end if; select * into j from public.faseca_journal where id = p; if not found or j.ancien is null then raise exception 'Version introuvable.'; end if; perform set_config('faseca.restore', '1', true); insert into public.faseca_inscriptions select * from jsonb_populate_record(null::public.faseca_inscriptions, j.ancien) on conflict (id) do update set numero = excluded.numero, organisation = excluded.organisation, responsable = excluded.responsable, telephone = excluded.telephone, commune = excluded.commune, email = excluded.email, secteurs = excluded.secteurs, produits = excluded.produits, details = excluded.details, formule = excluded.formule, mode_paiement = excluded.mode_paiement, statut = excluded.statut, montant_verse = excluded.montant_verse, stand = excluded.stand, notes = excluded.notes, photos = excluded.photos, updated_at = now(); perform set_config('faseca.restore', '', true); end $$; revoke execute on function public.faseca_restaurer(uuid) from public, anon; grant execute on function public.faseca_restaurer(uuid) to authenticated; -- 6. Un numéro de stand ne peut être attribué qu'une fois create unique index if not exists faseca_stand_unique on public.faseca_inscriptions (lower(btrim(stand))) where btrim(coalesce(stand,'')) <> '' and statut not in ('Refusée','Liste d''attente'); -- 7. Règles d'accès aux dossiers : un seul jeu clair drop policy if exists fi_select on public.faseca_inscriptions; drop policy if exists fi_insert on public.faseca_inscriptions; drop policy if exists fi_update on public.faseca_inscriptions; drop policy if exists fi_delete on public.faseca_inscriptions; drop policy if exists i_sel on public.faseca_inscriptions; drop policy if exists i_ins on public.faseca_inscriptions; drop policy if exists i_upd on public.faseca_inscriptions; drop policy if exists i_del on public.faseca_inscriptions; create policy i_sel on public.faseca_inscriptions for select to authenticated using (user_id = auth.uid() or public.is_staff()); create policy i_ins on public.faseca_inscriptions for insert to authenticated with check (user_id = auth.uid() and public.faseca_actif()); create policy i_upd on public.faseca_inscriptions for update to authenticated using ((user_id = auth.uid() and public.faseca_actif()) or public.is_super()) with check ((user_id = auth.uid() and public.faseca_actif()) or public.is_super()); create policy i_del on public.faseca_inscriptions for delete to authenticated using (public.is_super()); alter table public.faseca_support alter column user_id set default auth.uid(); commit; -- Contrôle : comptes disposant d'un accès équipe select email, role, actif from public.profiles where role <> 'porteur' union all select email, role || ' (liste équipe)', null from public.faseca_equipe order by 1;
