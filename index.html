TRIGGERS MÉTIER -- --------------------------------------------------------------------- create or replace function public.faseca_before_insert() returns trigger language plpgsql security definer set search_path = public as $$ declare cfg jsonb; lim timestamptz; maxs int; n int; begin if not public.is_super() then select data into cfg from public.faseca_config where id = 1; lim := nullif(cfg->>'limite','')::timestamptz; if lim is not null and now() > lim then raise exception 'Les inscriptions sont closes.'; end if; maxs := coalesce((cfg->>'stands')::int, 50); select count(*) into n from public.faseca_inscriptions where statut not in ('Refusée','Liste d''attente'); new.user_id := auth.uid(); new.email := lower(coalesce(auth.jwt()->>'email', new.email)); end if; return new; end $$; drop trigger if exists faseca_bi on public.faseca_inscriptions; create trigger faseca_bi before insert on public.faseca_inscriptions for each row execute function public.faseca_before_insert(); create or replace function public.faseca_before_update() returns trigger language plpgsql security definer set search_path = public as $$ begin if not public.is_super() then new.id := old.id; new.numero := old.numero; new.user_id := old.user_id; new.email := old.email; new.formule := old.formule; new.statut := old.statut; new.montant_verse := old.montant_verse; new.stand := old.stand; new.notes := old.notes; new.created_at := old.created_at; end if; return new; end $$; drop trigger if exists faseca_bu on public.faseca_inscriptions; create trigger faseca_bu before update on public.faseca_inscriptions for each row execute function public.faseca_before_update(); create or replace function public.faseca_log() returns trigger language plpgsql security definer set search_path = public as $$ declare act text := coalesce(auth.jwt()->>'email', 'système'); begin if tg_op = 'INSERT' then insert into public.faseca_journal(inscription_id, numero, operation, acteur, nouveau) values (new.id, new.numero, 'création', act, to_jsonb(new)); elsif tg_op = 'UPDATE' then insert into public.faseca_journal(inscription_id, numero, operation, acteur, ancien, nouveau) values (new.id, new.numero, 'modification', act, to_jsonb(old), to_jsonb(new)); else insert into public.faseca_journal(inscription_id, numero, operation, acteur, ancien) values (old.id, old.numero, 'suppression', act, to_jsonb(old)); end if; return null; end $$; drop trigger if exists faseca_aj on public.faseca_inscriptions; create trigger faseca_aj after insert or update or delete on public.faseca_inscriptions for each row execute function public.faseca_log(); create or replace function public.faseca_restaurer(p uuid) returns void language plpgsql security definer set search_path = public as $$ declare j public.faseca_journal; end $$; revoke all on function public.faseca_restaurer(uuid) from public, anon; grant execute on function public.faseca_restaurer(uuid) to authenticated; -- --------------------------------------------------------------------- -- 6.
